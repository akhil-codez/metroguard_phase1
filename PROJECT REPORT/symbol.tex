\chapter*{List of Symbols and Abbreviations}
\addcontentsline{toc}{chapter}{List of Symbols and Abbreviations}%

\newenvironment{symbols}

\begin{itemize} \setlength{\itemsep}{3pt}
	\item[\textbf{AI}] Artificial Intelligence
	\item[\textbf{CCTV}] Closed-Circuit Television
	\item[\textbf{CNN}] Convolutional Neural Network
	\item[\textbf{COCO}] Common Objects in Context
	\item[\textbf{CUDA}] Compute Unified Device Architecture
	\item[\textbf{DINO}] DETR with Improved Denoising Anchor Boxes
	\item[\textbf{FPS}] Frames Per Second
	\item[\textbf{GFLOPs}] Giga Floating-Point Operations
	\item[\textbf{GPU}] Graphics Processing Unit
	\item[\textbf{IoU}] Intersection over Union
	\item[\textbf{KTU}] APJ Abdul Kalam Technological University
	\item[\textbf{MAE}] Mean Absolute Error
	\item[\textbf{mAP}] Mean Average Precision
	\item[\textbf{MOT}] Multi-Object Tracking
	\item[\textbf{MSE}] Mean Signed Error
	\item[\textbf{NMS}] Non-Maximum Suppression
	\item[\textbf{OpenCV}] Open Source Computer Vision Library
	\item[\textbf{RGB}] Red, Green, Blue
	\item[\textbf{RMSE}] Root Mean Squared Error
	\item[\textbf{ROI}] Region of Interest
	\item[\textbf{SAM}] Segment Anything Model
	\item[\textbf{SOT}] Single Object Tracking
	\item[\textbf{TAM}] Target Tracking Module
	\item[\textbf{VAD}] Video Anomaly Detection
	\item[\textbf{WAPE}] Weighted Absolute Percentage Error
	\item[\textbf{YOLO}] You Only Look Once
	\item[$\rho_{img}$] Image-space normalized crowd density
	\item[$N_{ROI}$] Number of tracked pedestrians inside Region of Interest
	\item[$A_{ROI}$] Area of Region of Interest in megapixels
	\item[$\mathcal{B}_i$] Bounding box coordinates $(x_1, y_1, x_2, y_2)$ of detected person $i$
	\item[$\mathcal{F}_i$] Ground-contact footpoint $(x_{mid}, y_2)$ of pedestrian $i$
	\item[$\theta_{conf}$] Detection confidence threshold
	\item[$R_{incident}$] Composite incident risk score
\end{itemize}
